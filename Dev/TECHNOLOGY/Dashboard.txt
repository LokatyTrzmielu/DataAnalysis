const clamp = (x, a, b) => Math.max(a, Math.min(b, x));

function parseCsv(text) {
  // Expected headers (any order, case-insensitive):
  // sku, velocity_pctl (0-1 or 0-100), ratio, cube (LOW/HIGH), lines_p95(optional), cube_p95(optional)
  const lines = text
    .split(/\r?\n/)
    .map((l) => l.trim())
    .filter((l) => l.length > 0);
  if (lines.length < 2) return { rows: [], error: "CSV musi mieć nagłówek i co najmniej 1 wiersz danych." };

  const header = lines[0].split(",").map((h) => h.trim().toLowerCase());
  const idx = (name) => header.findIndex((h) => h === name);

  const iSku = idx("sku");
  const iVel = idx("velocity_pctl");
  const iRatio = idx("ratio");
  const iCube = idx("cube");
  const iLinesP95 = idx("lines_p95");
  const iCubeP95 = idx("cube_p95");

  if (iSku === -1 || iVel === -1 || iRatio === -1 || iCube === -1) {
    return {
      rows: [],
      error:
        "Brakuje wymaganych kolumn. Wymagane: sku, velocity_pctl, ratio, cube. Opcjonalne: lines_p95, cube_p95.",
    };
  }

  const rows = [];
  for (let i = 1; i < lines.length; i++) {
    const parts = lines[i].split(",").map((p) => p.trim());
    const sku = parts[iSku] ?? "";
    if (!sku) continue;

    let vp = Number(parts[iVel]);
    if (Number.isNaN(vp)) vp = 0;
    // allow 0-100 as input
    if (vp > 1) vp = vp / 100;
    vp = clamp(vp, 0, 1);

    const ratio = clamp(Number(parts[iRatio] ?? 0), 0, 999);
    const cubeRaw = (parts[iCube] ?? "LOW").toUpperCase();
    const cube = cubeRaw === "HIGH" ? "HIGH" : "LOW";

    const linesP95 = iLinesP95 !== -1 ? Number(parts[iLinesP95] ?? "") : undefined;
    const cubeP95 = iCubeP95 !== -1 ? Number(parts[iCubeP95] ?? "") : undefined;

    rows.push({ sku, velocity_pctl: vp, ratio, cube, lines_p95: linesP95, cube_p95: cubeP95 });
  }

  return { rows, error: "" };
}

function bandLabelVelocity(v, th) {
  if (v >= th.veryHigh) return "VERY_HIGH";
  if (v >= th.high) return "HIGH";
  if (v >= th.medium) return "MEDIUM";
  return "LOW";
}

function bandLabelRatio(r, th) {
  if (r > th.critical) return "CRITICAL";
  if (r > th.bad) return "BAD";
  if (r > th.risk) return "RISK";
  if (r >= th.ok) return "OK";
  return "FAST";
}

function loadClassFromRatio(r, th) {
  if (r < th.fast) return "LOW_LOAD";
  if (r < th.okUpper) return "MODERATE_LOAD";
  if (r < th.riskUpper) return "HIGH_LOAD";
  return "VERY_HIGH_LOAD";
}

function scoreTech({ velocityBand, loadBand, cube }, model) {
  const w = model.weights;
  const vScore = model.technologiesVelocity[velocityBand] ?? 0;
  const rScore = model.technologiesRatio[loadBand] ?? 0;
  const cScore = model.technologiesCube[cube] ?? 0;
  const base = vScore * w.velocity + rScore * w.ratio + cScore * w.cube;
  return clamp(base, 0, 100);
}

function computeAllScores(row, cfg) {
  const velocityBand = bandLabelVelocity(row.velocity_pctl, cfg.velocityThresholds);
  const segment = bandLabelRatio(row.ratio, cfg.ratioThresholds);

  // derive load band for scoring (LOW/MOD/HIGH/VERY_HIGH)
  const loadBand = loadClassFromRatio(row.ratio, {
    fast: cfg.ratioThresholds.fastUpper,
    okUpper: cfg.ratioThresholds.okUpper,
    riskUpper: cfg.ratioThresholds.riskUpper,
  });

  const baseScores = cfg.technologyModels.map((t) => {
    let s = scoreTech({ velocityBand, loadBand, cube: row.cube }, t.scoring);

    // post rules (simple & explainable)
    if (segment === "CRITICAL") {
      if (t.name !== "Manual" && t.name !== "Outside") s = 0;
    }
    if (segment === "BAD") {
      if (["MiB", "AutoStore", "Hai"].includes(t.name)) s = Math.max(0, s - 30);
    }
    if (row.cube === "HIGH") {
      if (t.name === "VLM") s = clamp(s + 10, 0, 100);
      if (t.name === "AutoStore") s = clamp(s + 5, 0, 100);
    }
    if (velocityBand === "LOW") {
      if (t.name === "Manual") s = clamp(s + 10, 0, 100);
    }

    return { technology: t.name, score: Math.round(s) };
  });

  baseScores.sort((a, b) => b.score - a.score);

  // Decision hint
  let decision = "CHECK";
  if (segment === "CRITICAL") decision = "OUTSIDE_CRITICAL";
  else if (segment === "BAD") decision = "OUTSIDE";
  else if (velocityBand === "LOW" && segment !== "FAST") decision = "OUTSIDE";
  else decision = baseScores[0]?.technology ?? "CHECK";

  return { velocityBand, segment, loadBand, decision, ranking: baseScores };
}

function heatColor(score) {
  // 0..100 → red..green (soft)
  const s = clamp(score, 0, 100) / 100;
  const r = Math.round(245 - 120 * s);
  const g = Math.round(110 + 120 * s);
  const b = Math.round(120 - 60 * s);
  return `rgb(${r}, ${g}, ${b})`;
}

function segBadge(seg) {
  const map = {
    FAST: "bg-emerald-600",
    OK: "bg-emerald-500",
    RISK: "bg-amber-500",
    BAD: "bg-red-500",
    CRITICAL: "bg-red-700",
  };
  return map[seg] ?? "bg-slate-500";
}

function velBadge(v) {
  const map = {
    VERY_HIGH: "bg-fuchsia-600",
    HIGH: "bg-indigo-600",
    MEDIUM: "bg-sky-600",
    LOW: "bg-slate-600",
  };
  return map[v] ?? "bg-slate-600";
}

export default function DashboardSkuTech() {

  const [cfg, setCfg] = useState(defaultConfig);
  const [activeSku, setActiveSku] = useState("SKU_001");

  const parsed = useMemo(() => parseCsv(csvText), [csvText]);

  const enriched = useMemo(() => {
    if (parsed.error) return [];
    return parsed.rows.map((r) => {
      const x = computeAllScores(r, cfg);
      return { ...r, ...x };
    });
  }, [parsed, cfg]);

  const skuList = useMemo(() => enriched.map((e) => e.sku), [enriched]);

  const selected = useMemo(() => enriched.find((e) => e.sku === activeSku) ?? enriched[0], [enriched, activeSku]);

  const overallRanking = useMemo(() => {
    // average score per technology across SKUs (only eligible ones)
    const techs = cfg.technologyModels.map((t) => t.name);
    const buckets = Object.fromEntries(techs.map((t) => [t, { sum: 0, n: 0 }]));
    enriched.forEach((e) => {
      // ignore critical/bad from averaging (they typically go outside)
      if (["CRITICAL"].includes(e.segment)) return;
      e.ranking.forEach((r) => {
        if (!buckets[r.technology]) return;
        buckets[r.technology].sum += r.score;
        buckets[r.technology].n += 1;
      });
    });
    const out = techs
      .map((t) => ({ technology: t, score: buckets[t].n ? Math.round(buckets[t].sum / buckets[t].n) : 0 }))
      .sort((a, b) => b.score - a.score);
    return out;
  }, [enriched, cfg.technologyModels]);

  const heatmap = useMemo(() => {
    // Heatmap: Velocity bands (rows) x Ratio segments (cols)
    const vBands = ["VERY_HIGH", "HIGH", "MEDIUM", "LOW"];
    const rSegs = ["FAST", "OK", "RISK", "BAD", "CRITICAL"];

    const cell = {};
    vBands.forEach((vb) => {
      rSegs.forEach((rs) => {
        cell[`${vb}|${rs}`] = { vb, rs, count: 0, topTech: "-", topScoreAvg: 0 };
      });
    });

    // for each SKU, contribute to its cell based on its top recommendation score
    enriched.forEach((e) => {
      const key = `${e.velocityBand}|${e.segment}`;
      const best = e.ranking?.[0];
      if (!cell[key] || !best) return;
      cell[key].count += 1;
      // accumulate best score to average
      cell[key].topScoreAvg += best.score;
      // keep most frequent best technology (simple)
      cell[key].topTech = best.technology;
    });

    const grid = vBands.map((vb) =>
      rSegs.map((rs) => {
        const c = cell[`${vb}|${rs}`];
        const avg = c.count ? Math.round(c.topScoreAvg / c.count) : 0;
        return { ...c, topScoreAvg: avg };
      })
    );

    return { vBands, rSegs, grid };
  }, [enriched]);

  const techBarData = useMemo(() => {
    // For selected SKU: show top 6 tech scores
    if (!selected?.ranking) return [];
    return selected.ranking.slice(0, 6).map((d) => ({ name: d.technology, score: d.score }));
  }, [selected]);

  const segmentSummary = useMemo(() => {
    const counts = { FAST: 0, OK: 0, RISK: 0, BAD: 0, CRITICAL: 0 };
    enriched.forEach((e) => {
      counts[e.segment] = (counts[e.segment] ?? 0) + 1;
    });
    return Object.entries(counts).map(([k, v]) => ({ segment: k, count: v }));
  }, [enriched]);

  const helperText = (
    <Alert className="bg-slate-50">
      <Info className="h-4 w-4" />
      <AlertTitle>Jak używać</AlertTitle>
      <AlertDescription className="text-sm leading-relaxed">
        Wklej CSV (nagłówki: <span className="font-mono">sku,velocity_pctl,ratio,cube</span>). Następnie zobacz:
        <ul className="list-disc pl-5 mt-2">
          <li><b>Heatmap</b>: rozkład SKU po Velocity vs Segment (ratio) i średni „best score”.</li>
          <li><b>Ranking</b>: top technologie dla wybranego SKU + ranking globalny (średnia po SKU).</li>
          <li><b>Settings</b>: progi Kardex oraz wagi scoringu.</li>
        </ul>
      </AlertDescription>
    </Alert>
  );

  return (
    <div className="p-6 space-y-6">
      <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.35 }}>
        <div className="flex items-start justify-between gap-4 flex-wrap">
          <div>
            <h1 className="text-2xl font-semibold tracking-tight">SKU Technology Dashboard</h1>
            <p className="text-sm text-muted-foreground mt-1">
              Heatmap (Velocity × Ratio Segment) + ranking technologii (scoring 0–100). Prototyp do Twojej aplikacji.
            </p>
          </div>
          <div className="flex gap-2">
            <Button variant="secondary" onClick={() => setCsvText(defaultCsv)}>
              <Wand2 className="h-4 w-4 mr-2" /> Załaduj przykładowe dane
            </Button>
          </div>
        </div>
      </motion.div>

      {helperText}

      {parsed.error ? (
        <Alert variant="destructive">
          <AlertTitle>Błąd CSV</AlertTitle>
          <AlertDescription>{parsed.error}</AlertDescription>
        </Alert>
      ) : null}

      <Tabs defaultValue="overview" className="w-full">
        <TabsList className="flex flex-wrap">
          <TabsTrigger value="overview">Overview</TabsTrigger>
          <TabsTrigger value="heatmap">Heatmap</TabsTrigger>
          <TabsTrigger value="ranking">Ranking</TabsTrigger>

                    <b>FAST/OK</b> oznacza, że replenishment ratio jest w granicach akceptowalnych dla automatyki.
                    <b> RISK/BAD/CRITICAL</b> to strefa problemowa (częste zasilenia, korki, spadek throughput).
                  </p>
